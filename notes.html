1. Introduction to Arrays

An array is a collection of elements of the same data type stored in contiguous memory locations. Arrays are one of the most fundamental data structures in computer science and are used extensively in algorithm design and problem solving.

1.1 Array Declaration and Initialization

  • Static Array: Size is fixed at compile time - int arr[10];
  • Dynamic Array: Size is determined at runtime - int* arr = new int[n];
  • Initialization: int arr[5] = {1, 2, 3, 4, 5};
// Array Declaration
int arr[5]; // Uninitialized
int arr[5] = {0}; // All elements = 0
int arr[] = {1,2,3,4,5}; // Auto size

3. Key Algorithms

3.1 Linear Search

Linear search is the simplest searching algorithm. It compares the search element with each element in the array one by one.

int linearSearch(int arr[], int n, int target) {
  for (int i = 0; i < n; i++) {
    if (arr[i] == target) return i;
  }
  return -1; // Not found
}
// Time: O(n), Space: O(1)

3.2 Binary Search

Binary search works on sorted arrays. It divides the search space in half each time, resulting in logarithmic time complexity.

int binarySearch(int arr[], int n, int target) {
  int left = 0, right = n - 1;
  while (left <= right) {
    int mid = left + (right - left) / 2;
    if (arr[mid] == target) return mid;
    if (arr[mid] < target) left = mid + 1;
    else right = mid - 1;
  }
  return -1;
}
// Time: O(log n), Space: O(1)

3.3 Reverse Array

Reversing an array can be done efficiently using the two-pointer technique without needing extra space.

void reverseArray(int arr[], int n) {
  int left = 0, right = n - 1;
  while (left < right) {
    int temp = arr[left];
    arr[left] = arr[right];
    arr[right] = temp;
    left++; right--;
  }
}
// Time: O(n), Space: O(1)
4. Common Interview Problems

4.1 Two Sum Problem

Find two numbers that add up to target. Use HashMap for O(n) solution.

vector<int> twoSum(vector<int>& nums, int target) {
  unordered_map<int, int> map; // {value: index}
  
  for (int i = 0; i < nums.size(); i++) {
    int complement = target - nums[i];
    
    if (map.find(complement) != map.end()) {
      return {map[complement], i};
    }
    map[nums[i]] = i;
  }
  return {};
}
// Time: O(n), Space: O(n)

4.2 Max Subarray Sum (Kadane's Algorithm)

Find maximum sum of contiguous subarray using dynamic programming.

int maxSubArray(vector<int>& nums) {
  int maxSum = nums[0];
  int currentSum = nums[0];
  
  for (int i = 1; i < nums.size(); i++) {
    currentSum = max(nums[i], currentSum + nums[i]);
    maxSum = max(maxSum, currentSum);
  }
  return maxSum;
}
// Example: [-2,1,-3,4,-1,2,1,-5,4] → 6 (subarray [4,-1,2,1])
// Time: O(n), Space: O(1)

4.3 Container with Most Water

Two-pointer approach to find max area between two lines.

int maxArea(vector<int>& height) {
  int maxArea = 0;
  int left = 0, right = height.size() - 1;
  
  while (left < right) {
    int width = right - left;
    int h = min(height[left], height[right]);
    int area = width * h;
    
    maxArea = max(maxArea, area);
    
    // Move pointer pointing to smaller height
    if (height[left] < height[right]) left++;
    else right--;
  }
  return maxArea;
}
// Time: O(n), Space: O(1)

4.4 Majority Element

Find element appearing more than n/2 times using Boyer-Moore voting.

int majorityElement(vector<int>& nums) {
  int candidate = 0, count = 0;
  
  // Find candidate
  for (int num : nums) {
    if (count == 0) candidate = num;
    count += (num == candidate) ? 1 : -1;
  }
  
  return candidate;
}
// Time: O(n), Space: O(1)

4.5 Best Time to Buy Stock

Single pass to track minimum price and maximum profit.

int maxProfit(vector<int>& prices) {
  int minPrice = INT_MAX;
  int maxProfit = 0;
  
  for (int price : prices) {
    minPrice = min(minPrice, price);
    maxProfit = max(maxProfit, price - minPrice);
  }
  return maxProfit;
}
// Example: [7,1,5,3,6,4] → 5 (buy at 1, sell at 6)
// Time: O(n), Space: O(1)

1. String Fundamentals

Strings are sequences of characters. In many languages like Java and Python, strings are immutable, meaning once created, they cannot be changed. Understanding string properties is crucial for solving string-related problems efficiently.

1.1 String Properties

  • Immutability: In Java/Python, strings cannot be modified after creation
  • Character Encoding: ASCII (0-127), Extended ASCII (0-255), Unicode (0-1,114,111)
  • String Comparison: Use equals() method, not == operator
  • Substring: String.substring(start, end) extracts portion of string

s.substring(1, 4) // Returns "ell"
s.indexOf("l") // Returns 2
s.toUpperCase() // Returns "HELLO"

3. Essential Techniques

3.1 Two-Pointer Approach

Used to solve problems involving palindromes, reversal, and matching pairs in strings.

bool isPalindrome(string s) {
  int left = 0, right = s.length() - 1;
  while (left < right) {
    if (s[left] != s[right]) return false;
    left++; right--;
  }
  return true;
}
// Time: O(n), Space: O(1)

3.2 Sliding Window

Efficient technique for finding substrings with specific properties (like longest substring without repeating characters).

int lengthOfLongestSubstring(string s) {
  unordered_map map;
  int left = 0, maxLen = 0;
  for (int right = 0; right < s.length(); right++) {
    if (map.find(s[right]) != map.end()) {
      left = max(left, map[s[right]] + 1);
    }
    map[s[right]] = right;
    maxLen = max(maxLen, right - left + 1);
  }
  return maxLen;
}
// Time: O(n), Space: O(min(m,n))

3.3 Character Frequency Counting

Common technique using HashMap or Array to count character occurrences for anagram checking, etc.

bool isAnagram(string s, string t) {
  if (s.length() != t.length()) return false;
  int freq[26] = {0};
  for (char c : s) freq[c - 'a']++;
  for (char c : t) {
    if (--freq[c - 'a'] < 0) return false;
  }
  return true;
}
// Time: O(n), Space: O(1) - fixed 26 letters
4. Important Interview Problems

4.1 Reverse String

Reverse array of characters in-place using two pointers.

void reverseString(vector<char>& s) {
  int left = 0, right = s.size() - 1;
  
  while (left < right) {
    swap(s[left], s[right]);
    left++;
    right--;
  }
}
// Time: O(n), Space: O(1)

4.2 Valid Palindrome

Check if string is palindrome, ignoring spaces and special characters.

bool isPalindrome(string s) {
  int left = 0, right = s.length() - 1;
  
  while (left < right) {
    while (left < right && !isalnum(s[left])) left++;
    while (left < right && !isalnum(s[right])) right--;
    
    if (tolower(s[left]) != tolower(s[right]))
      return false;
    
    left++;
    right--;
  }
  return true;
}
// Time: O(n), Space: O(1)

4.3 Longest Substring Without Repeating

Find longest substring with all unique characters using sliding window.

int lengthOfLongestSubstring(string s) {
  unordered_map<char, int> map;
  int left = 0, maxLen = 0;
  
  for (int right = 0; right < s.length(); right++) {
    if (map.find(s[right]) != map.end()) {
      left = max(left, map[s[right]] + 1);
    }
    map[s[right]] = right;
    maxLen = max(maxLen, right - left + 1);
  }
  return maxLen;
}
// Example: "abcabcbb" → 3 ("abc")
// Time: O(n), Space: O(min(m,n))

4.4 Group Anagrams

Group words that are anagrams of each other.

vector<vector<string>> groupAnagrams(vector<string>& strs) {
  unordered_map<string, vector<string>> map;
  
  for (string s : strs) {
    string sorted_s = s;
    sort(sorted_s.begin(), sorted_s.end());
    map[sorted_s].push_back(s);
  }
  
  vector<vector<string>> result;
  for (auto& p : map) {
    result.push_back(p.second);
  }
  return result;
}
// Time: O(n*k*log k), Space: O(n*k) [k=avg string length]

1. Linked List Fundamentals

A linked list is a linear data structure where elements are stored in nodes. Each node contains data and a reference (link) to the next node. Unlike arrays, linked lists don't require contiguous memory and support dynamic resizing.

1.1 Node Structure

struct Node {
  int data;
  Node* next;
  
  Node(int val) {
    data = val;
    next = NULL;
  }
}

1.2 Types of Linked Lists

  • Singly Linked List: Each node points to the next node only
  • Doubly Linked List: Each node points to both next and previous nodes
  • Circular Linked List: Last node points back to the first node
2. Time & Space Complexity
OperationTimeSpace
Access element at position iO(n)O(1)
Insert at beginningO(1)O(1)
Delete from beginningO(1)O(1)
Insert/Delete at position iO(n)O(1)
3. Essential Operations

3.1 Reverse Linked List (Iterative)

Reverse the entire linked list by changing the direction of all pointers.

Node* reverseList(Node* head) {
  Node* prev = NULL, *curr = head;
  while (curr) {
    Node* temp = curr->next; // Save next
    curr->next = prev; // Reverse link
    prev = curr; // Move prev
    curr = temp; // Move curr
  }
  return prev; // New head
}
// Time: O(n), Space: O(1)

3.2 Detect Cycle (Floyd's Algorithm)

Use slow and fast pointers to detect if a cycle exists in the linked list.

bool hasCycle(Node* head) {
  if (!head || !head->next) return false;
  Node* slow = head, *fast = head->next;
  while (fast && fast->next) {
    if (slow == fast) return true; // Cycle found
    slow = slow->next; // Move by 1
    fast = fast->next->next; // Move by 2
  }
  return false;
}
// Time: O(n), Space: O(1)

3.3 Find Middle Element

Use slow and fast pointers to find the middle element in one pass.

Node* findMiddle(Node* head) {
  Node* slow = head, *fast = head;
  while (fast && fast->next) {
    slow = slow->next; // Move by 1
    fast = fast->next->next; // Move by 2
  }
  return slow; // Middle element
}
// Time: O(n), Space: O(1)
4. Advanced Problems

4.1 Merge Two Sorted Linked Lists

Combine two sorted lists maintaining order.

ListNode* mergeTwoLists(ListNode* l1, ListNode* l2) {
  ListNode dummy(0);
  ListNode* curr = &dummy;
  
  while (l1 && l2) {
    if (l1->val < l2->val) {
      curr->next = l1;
      l1 = l1->next;
    } else {
      curr->next = l2;
      l2 = l2->next;
    }
    curr = curr->next;
  }
  
  curr->next = l1 ? l1 : l2;
  return dummy.next;
}
// Time: O(m+n), Space: O(1)

4.2 Palindrome Linked List

Check if linked list is a palindrome using slow/fast pointers.

bool isPalindrome(ListNode* head) {
  if (!head || !head->next) return true;
  
  // Find middle
  ListNode* slow = head, *fast = head;
  while (fast && fast->next) {
    slow = slow->next;
    fast = fast->next->next;
  }
  
  // Reverse second half
  ListNode* prev = NULL, *curr = slow;
  while (curr) {
    ListNode* temp = curr->next;
    curr->next = prev;
    prev = curr;
    curr = temp;
  }
  
  // Compare
  while (prev) {
    if (head->val != prev->val) return false;
    head = head->next;
    prev = prev->next;
  }
  return true;
}
// Time: O(n), Space: O(1)

4.3 Remove Nth Node From End

Remove the nth node from the end of list using two pointers.

ListNode* removeNthFromEnd(ListNode* head, int n) {
  ListNode dummy(0);
  dummy.next = head;
  ListNode* first = &dummy, *second = &dummy;
  
  // Move first pointer n+1 steps ahead
  for (int i = 0; i <= n; i++) {
    first = first->next;
  }
  
  // Move both until first reaches end
  while (first) {
    first = first->next;
    second = second->next;
  }
  
  second->next = second->next->next;
  return dummy.next;
}
// Time: O(n), Space: O(1)

4.4 LRU Cache Implementation

Implement using doubly linked list and HashMap for O(1) access and update.

class LRUCache {
private:
  unordered_map<int, list<pair<int,int>>::iterator> map;
  list<pair<int,int>> dq; // {key: value}
  int capacity;
public:
  LRUCache(int cap) : capacity(cap) {}
  
  int get(int key) {
    if (map.find(key) == map.end()) return -1;
    dq.splice(dq.begin(), dq, map[key]);
    return map[key]->second;
  }
  
  void put(int key, int value) {
    if (map.find(key) != map.end()) {
      map[key]->second = value;
      dq.splice(dq.begin(), dq, map[key]);
      return;
    }
    if (map.size() == capacity) {
      map.erase(dq.back().first);
      dq.pop_back();
    }
    dq.push_front({key, value});
    map[key] = dq.begin();
  }
};
// Time: O(1) for both get and put

1. Stack Fundamentals

A stack is a linear data structure that follows the LIFO (Last In First Out) principle. The last element added is the first one to be removed. Stacks are widely used in function calls, expression evaluation, and backtracking algorithms.

1.1 Core Operations

  • Push: Add an element to the top - O(1)
  • Pop: Remove and return top element - O(1)
  • Peek/Top: View top element without removing - O(1)
  • isEmpty: Check if stack is empty - O(1)
  • Size: Get number of elements - O(1)
template
class Stack {
private:
  vector data;
public:
  void push(T val) { data.push_back(val); }
  void pop() { if (!data.empty()) data.pop_back(); }
  T top() { return data.back(); }
  bool isEmpty() { return data.empty(); }
}
2. Applications of Stack
  • Function Call Stack: OS uses stack to manage function calls and returns
  • Expression Evaluation: Convert infix to postfix, evaluate postfix expressions
  • Backtracking: DFS, Maze solving, N-Queens problem
  • Undo/Redo Functionality: Common in text editors
  • Browser History: Back button uses stack
3. Important Problems

3.1 Valid Parentheses

Check if parentheses, brackets, and braces are balanced and properly nested.

bool isValid(string s) {
  stack st;
  map match = {{')', '('}, {']', '['}, {'}', '{'}};
  
  for (char c : s) {
    if (c == '(' || c == '[' || c == '{') {
      st.push(c);
    } else {
      if (st.empty() || st.top() != match[c])
        return false;
      st.pop();
    }
  }
  return st.empty();
}
// Time: O(n), Space: O(n)

3.2 Next Greater Element (NGE)

For each element, find the nearest element to its right that is greater than it.

vector nextGreater(vector& arr) {
  int n = arr.size();
  vector result(n, -1);
  stack st; // Store indices
  
  for (int i = n - 1; i >= 0; i--) {
    while (!st.empty() && arr[st.top()] <= arr[i]) {
      st.pop();
    }
    if (!st.empty()) result[i] = arr[st.top()];
    st.push(i);
  }
  return result;
}
// Time: O(n), Space: O(n)

3.3 Min Stack

Implement a stack that supports push, pop, top, and retrieving minimum in O(1) time.

class MinStack {
private:
  stack st, minSt;
public:
  void push(int x) {
    st.push(x);
    if (minSt.empty() || x <= minSt.top())
      minSt.push(x);
  }
  void pop() {
    if (st.top() == minSt.top()) minSt.pop();
    st.pop();
  }
  int getMin() { return minSt.top(); }
}

4.3 Trapping Rain Water

Calculate total water trapped between elevation bars.

int trap(vector<int>& height) {
  if (height.empty()) return 0;
  
  vector<int> leftMax(height.size());
  vector<int> rightMax(height.size());
  
  leftMax[0] = height[0];
  for (int i = 1; i < height.size(); i++) {
    leftMax[i] = max(leftMax[i-1], height[i]);
  }
  
  rightMax[height.size()-1] = height[height.size()-1];
  for (int i = height.size()-2; i >= 0; i--) {
    rightMax[i] = max(rightMax[i+1], height[i]);
  }
  
  int water = 0;
  for (int i = 0; i < height.size(); i++) {
    int minHeight = min(leftMax[i], rightMax[i]);
    water += minHeight - height[i];
  }
  return water;
}
// Time: O(n), Space: O(n)

1. Queue Fundamentals

A queue is a linear data structure that follows the FIFO (First In First Out) principle. The first element added is the first one to be removed. Queues are used in BFS, job scheduling, and resource allocation.

1.1 Core Operations

  • Enqueue: Add element to rear - O(1)
  • Dequeue: Remove from front - O(1)
  • Peek/Front: View front element - O(1)
  • isEmpty: Check if queue is empty - O(1)
  • Size: Get number of elements - O(1)
class Queue {
private:
  vector data;
  int front = 0;
public:
  void enqueue(int x) { data.push_back(x); }
  int dequeue() { return data[front++]; }
  int peek() { return data[front]; }
  bool isEmpty() { return front == data.size(); }
}
2. Types of Queues

2.1 Simple Queue

Elements are added at rear and removed from front. FIFO principle.

2.2 Circular Queue

Circular arrangement saves space. When rear reaches end, it wraps around to beginning if space is available.

2.3 Deque (Double-Ended Queue)

Elements can be added or removed from both ends. Combines properties of stack and queue.

deque dq;
dq.push_front(1); // Add to front
dq.push_back(2); // Add to back
dq.pop_front(); // Remove from front
dq.pop_back(); // Remove from back
dq.front(); // Access front
dq.back(); // Access back

2.4 Priority Queue

Elements are processed based on priority rather than order of arrival. Usually implemented using heap.

3. Key Applications
  • BFS (Breadth-First Search): Traverse graph level by level
  • Level Order Tree Traversal: Visit all nodes at same depth
  • Job Scheduling: CPU scheduling, process queues
  • Sliding Window Maximum: Using deque for efficient window tracking
  • LRU Cache: Deque + HashMap for least recently used items
4. Important Interview Problems

4.1 Implement Queue using Stacks

Create queue operations using two stacks.

class MyQueue {
private:
  stack<int> s1, s2;
public:
  void push(int x) {
    s1.push(x);
  }
  int pop() {
    if (s2.empty()) {
      while (!s1.empty()) {
        s2.push(s1.top());
        s1.pop();
      }
    }
    int val = s2.top();
    s2.pop();
    return val;
  }
};
// Push: O(1), Pop: O(1) amortized

4.2 Sliding Window Maximum

Find maximum in every sliding window using deque.

vector<int> maxSlidingWindow(vector<int>& nums, int k) {
  deque<int> dq; // Store indices
  vector<int> result;
  
  for (int i = 0; i < nums.size(); i++) {
    // Remove elements outside window
    if (!dq.empty() && dq.front() < i - k + 1) {
      dq.pop_front();
    }
    
    // Remove smaller elements
    while (!dq.empty() && nums[dq.back()] < nums[i]) {
      dq.pop_back();
    }
    
    dq.push_back(i);
    
    if (i >= k - 1) {
      result.push_back(nums[dq.front()]);
    }
  }
  return result;
}
// Time: O(n), Space: O(k)

4.3 Rotten Oranges

Find minimum time to rot all fresh oranges using BFS.

int orangesRotting(vector<vector<int>>& grid) {
  queue<pair<int,int>> q;
  int fresh = 0, time = 0;
  int dirs[4][2] = {{0,1}, {0,-1}, {1,0}, {-1,0}};
  
  for (int i = 0; i < grid.size(); i++) {
    for (int j = 0; j < grid[0].size(); j++) {
      if (grid[i][j] == 2) q.push({i, j});
      else if (grid[i][j] == 1) fresh++;
    }
  }
  
  while (!q.empty() && fresh) {
    int size = q.size();
    for (int i = 0; i < size; i++) {
      auto [x, y] = q.front();
      q.pop();
      for (auto& d : dirs) {
        int nx = x + d[0], ny = y + d[1];
        if (nx >= 0 && ny >= 0 && nx < grid.size() &&
            ny < grid[0].size() && grid[nx][ny] == 1) {
          grid[nx][ny] = 2;
          q.push({nx, ny});
          fresh--;
        }
      }
    }
    time++;
  }
  
  return fresh == 0 ? time : -1;
}
// Time: O(m*n), Space: O(m*n)

1. Tree Fundamentals

A tree is a non-linear hierarchical data structure where each element is called a node. The topmost node is the root, and nodes directly connected are parent and child nodes. Trees are essential for many applications including file systems, databases, and expression parsing.

1.1 Tree Terminology

  • Root: Top node with no parent
  • Parent-Child: Relationship between connected nodes
  • Leaf Node: Node with no children
  • Height: Maximum distance from root to leaf
  • Depth: Distance of node from root
  • Subtree: A tree formed by a node and its descendants
struct TreeNode {
  int val;
  TreeNode* left;
  TreeNode* right;
  
  TreeNode(int x) : val(x), left(NULL), right(NULL) {}
}
2. Tree Traversals

2.1 Depth-First Traversals (DFS)

  • Inorder (L-N-R): Left subtree → Node → Right subtree (gives sorted order in BST)
  • Preorder (N-L-R): Node → Left subtree → Right subtree (copy tree, create expression tree)
  • Postorder (L-R-N): Left subtree → Right subtree → Node (delete tree, evaluate postfix)
// Inorder Traversal
void inorder(TreeNode* root) {
  if (!root) return;
  inorder(root->left);
  cout << root->val << " ";
  inorder(root->right);
}
// Time: O(n), Space: O(h)

2.2 Breadth-First Traversal (BFS)

Level-order traversal using queue. Visits nodes level by level from top to bottom.

vector> levelOrder(TreeNode* root) {
  vector> result;
  if (!root) return result;
  
  queue q;
  q.push(root);
  
  while (!q.empty()) {
    int size = q.size();
    vector level;
    
    for (int i = 0; i < size; i++) {
      TreeNode* node = q.front();
      q.pop();
      level.push_back(node->val);
      
      if (node->left) q.push(node->left);
      if (node->right) q.push(node->right);
    }
    result.push_back(level);
  }
  return result;
}
3. Binary Search Tree (BST)

A BST is a binary tree where for each node, all values in left subtree are smaller and all values in right subtree are larger. This property enables efficient searching, insertion, and deletion.

3.1 BST Operations

  • Search: Compare with current node, go left if smaller, right if larger - O(log n) average, O(n) worst
  • Insert: Find correct position using BST property - O(log n) average, O(n) worst
  • Delete: Handle 3 cases: no children, one child, two children - O(log n) average
// BST Search
TreeNode* search(TreeNode* root, int val) {
  if (!root) return NULL;
  if (root->val == val) return root;
  if (val < root->val) return search(root->left, val);
  return search(root->right, val);
}
// Time: O(log n) avg, O(n) worst

4.2 Balanced Binary Tree

Check if left and right subtrees differ by at most 1 in height.

bool isBalanced(TreeNode* root) {

}

int getHeight(TreeNode* root) {
  if (!root) return 0;
  
  int left = getHeight(root->left);
  if (left == -1) return -1;
  
  int right = getHeight(root->right);
  if (right == -1) return -1;
  
  if (abs(left - right) > 1) return -1;
  return 1 + max(left, right);
}

4.3 Path Sum

Check if root-to-leaf path exists with given sum.

bool hasPathSum(TreeNode* root, int sum) {
  if (!root) return false;
  if (!root->left && !root->right && root->val == sum) return true;
  
  return hasPathSum(root->left, sum - root->val) ||
          hasPathSum(root->right, sum - root->val);
}
// Time: O(n), Space: O(h)

4.4 Lowest Common Ancestor (LCA)

Find lowest common ancestor of two nodes in BST.

TreeNode* lowestCommonAncestor(TreeNode* root, TreeNode* p, TreeNode* q) {
  if (!root) return NULL;
  
  // Both p and q are in left subtree
  if (p->val < root->val && q->val < root->val)
    return lowestCommonAncestor(root->left, p, q);
  
  // Both p and q are in right subtree
  if (p->val > root->val && q->val > root->val)
    return lowestCommonAncestor(root->right, p, q);
  
  // One on left, one on right (or one is root)
  return root;
}
// Time: O(log n) avg, O(n) worst; Space: O(h)

4.5 Binary Tree Maximum Path Sum

Find maximum path sum between any two nodes.

int maxPathSum(TreeNode* root) {
  int maxSum = INT_MIN;
  maxPathHelper(root, maxSum);
  return maxSum;
}

int maxPathHelper(TreeNode* root, int& maxSum) {
  if (!root) return 0;
  
  int left = max(0, maxPathHelper(root->left, maxSum));
  int right = max(0, maxPathHelper(root->right, maxSum));
  
  maxSum = max(maxSum, left + right + root->val);
  return root->val + max(left, right);
}
// Time: O(n), Space: O(h)

1. Graph Fundamentals

A graph is a non-linear data structure consisting of vertices (nodes) and edges (connections). Graphs can be directed or undirected, weighted or unweighted, and are used to represent relationships in social networks, maps, web pages, and more.

1.1 Graph Representations

  }

2. Graph Traversal Algorithms

2.1 Depth-First Search (DFS)

Traverses deeply into the graph before backtracking. Uses stack (implicit or explicit). Applications: cycle detection, topological sort, connected components.

void dfs(int node, vector>& adj, vector& visited) {
  visited[node] = true;
  cout << node << " ";
  
  for (int neighbor : adj[node]) {
    if (!visited[neighbor]) {
      dfs(neighbor, adj, visited);
    }
  }
}
// Time: O(V+E), Space: O(V)

2.2 Breadth-First Search (BFS)

Traverses level by level using queue. Applications: shortest path, word ladder, knight's move, level-order tree traversal.

void bfs(int start, vector>& adj) {
  vector visited(adj.size(), false);
  queue q;
  q.push(start);
  visited[start] = true;
  
  while (!q.empty()) {
    int node = q.front();
    q.pop();
    cout << node << " ";
    
    for (int neighbor : adj[node]) {
      if (!visited[neighbor]) {
        visited[neighbor] = true;
        q.push(neighbor);
      }
    }
  }
}
// Time: O(V+E), Space: O(V)

3.2 Topological Sorting

4.1 Number of Islands

Count number of connected components using DFS.

int numIslands(vector<vector<char>>& grid) {
  if (grid.empty()) return 0;
  int count = 0;
  int m = grid.size(), n = grid[0].size();
  
  for (int i = 0; i < m; i++) {
    for (int j = 0; j < n; j++) {
      if (grid[i][j] == '1') {
        dfs(grid, i, j);
        count++;
      }
    }
  }
  return count;
}

void dfs(vector<vector<char>>& grid, int i, int j) {
  if (i < 0 || i >= grid.size() || j < 0 || j >= grid[0].size()
      || grid[i][j] != '1') return;
  grid[i][j] = '0';
  dfs(grid, i+1, j);
  dfs(grid, i-1, j);
  dfs(grid, i, j+1);
  dfs(grid, i, j-1);
}
// Time: O(m*n), Space: O(m*n)

4.2 Course Schedule (Topological Sort)

Check if all courses can be completed given prerequisites.

bool canFinish(int n, vector<vector<int>>& edges) {
  vector<vector<int>> adj(n);
  vector<int> indegree(n, 0);
  
  for (auto& edge : edges) {
    adj[edge[1]].push_back(edge[0]);
    indegree[edge[0]]++;
  }
  
  queue<int> q;
  for (int i = 0; i < n; i++) {
    if (indegree[i] == 0) q.push(i);
  }
  
  int count = 0;
  while (!q.empty()) {
    int u = q.front();
    q.pop();
    count++;
    
    for (int v : adj[u]) {
      if (--indegree[v] == 0) q.push(v);
    }
  }
  return count == n;
}
// Time: O(V+E), Space: O(V+E)

Node* dfs(Node* node, unordered_map<Node*, Node*>& map) {
  if (map.find(node) != map.end()) return map[node];
  
  Node* clone = new Node(node->val);
  map[node] = clone;

4.4 Word Ladder (BFS Shortest Path)

Find shortest path between two words with one letter change.

int ladderLength(string begin, string end, vector<string>& words) {
  unordered_set<string> dict(words.begin(), words.end());
  queue<pair<string, int>> q;
  q.push({begin, 1});
  
  while (!q.empty()) {
    auto [word, level] = q.front();
    q.pop();
    
    if (word == end) return level;
    
    for (int i = 0; i < word.size(); i++) {
      for (char c = 'a'; c <= 'z'; c++) {
        string next = word;
        next[i] = c;
        if (dict.count(next)) {
          dict.erase(next);
          q.push({next, level + 1});
        }
      }
    }
  }
  return 0;
}
// Time: O(n*l*26), Space: O(n) [n=words, l=length]

1. Dynamic Programming Fundamentals

Dynamic Programming (DP) is an optimization technique for solving problems that have overlapping subproblems and optimal substructure. Instead of solving the same subproblems repeatedly, we store (memoize) their solutions.

1.1 Two Key Properties of DP Problems

  • Overlapping Subproblems: Problem can be broken down into subproblems which are reused several times
  • Optimal Substructure: Optimal solution contains optimal solutions to subproblems

1.2 DP Approaches

  • Memoization (Top-Down): Recursive approach with caching results
  • Tabulation (Bottom-Up): Iterative approach building solutions from base cases
// Fibonacci - Memoization Approach
map memo;

long long fib(int n) {
  if (n <= 1) return n;
  if (memo.find(n) != memo.end()) return memo[n];
  return memo[n] = fib(n-1) + fib(n-2);
}
// Time: O(n), Space: O(n)
// Fibonacci - Tabulation Approach
long long fib(int n) {
  if (n <= 1) return n;
  vector dp(n+1);
  dp[0] = 0, dp[1] = 1;
  
  for (int i = 2; i <= n; i++) {
    dp[i] = dp[i-1] + dp[i-2];
  }
  return dp[n];
}
// Time: O(n), Space: O(n)
2. Types of DP Problems

2.1 1D Dynamic Programming

  • Climbing Stairs: dp[i] = dp[i-1] + dp[i-2]
  • House Robber: dp[i] = max(dp[i-1], arr[i] + dp[i-2])
  • Coin Change: Minimum coins to make amount
// House Robber
int rob(vector& nums) {
  if (nums.empty()) return 0;
  if (nums.size() == 1) return nums[0];
  
  int prev1 = 0, prev2 = 0;
  
  for (int num : nums) {
    int curr = max(num + prev2, prev1);
    prev2 = prev1;
    prev1 = curr;
  }
  return prev1;
}
// Time: O(n), Space: O(1)

2.2 2D Dynamic Programming

  • Longest Common Subsequence (LCS): dp[i][j] = length of LCS of first i and j chars
  • Edit Distance (Levenshtein): Minimum operations to transform string
  • Unique Paths: Count ways to reach bottom-right in grid
  • Knapsack Problem: Maximum value with weight constraint
// Longest Common Subsequence
int lcs(string a, string b) {
  int m = a.size(), n = b.size();
  vector> dp(m+1, vector(n+1, 0));
  
  for (int i = 1; i <= m; i++) {
    for (int j = 1; j <= n; j++) {
      if (a[i-1] == b[j-1]) {
        dp[i][j] = dp[i-1][j-1] + 1;
      } else {
        dp[i][j] = max(dp[i-1][j], dp[i][j-1]);
      }
    }
  }
  return dp[m][n];
}
// Time: O(m*n), Space: O(m*n)

4.3 Longest Increasing Subsequence (LIS)

Find length of longest strictly increasing subsequence.

int lengthOfLIS(vector<int>& nums) {
  vector<int> dp;
  
  for (int num : nums) {
    auto it = lower_bound(dp.begin(), dp.end(), num);
    if (it == dp.end()) {
      dp.push_back(num);
    } else {
      *it = num;
    }
  }
  return dp.size();
}

4.4 Edit Distance (Levenshtein)

Minimum operations (insert, delete, replace) to transform s1 to s2.

int editDistance(string s1, string s2) {
  int m = s1.size(), n = s2.size();
  vector<vector<int>> dp(m+1, vector<int>(n+1));
  
  for (int i = 0; i <= m; i++) dp[i][0] = i;
  for (int j = 0; j <= n; j++) dp[0][j] = j;
  
  for (int i = 1; i <= m; i++) {
    for (int j = 1; j <= n; j++) {
      if (s1[i-1] == s2[j-1]) {
        dp[i][j] = dp[i-1][j-1];
      } else {
        dp[i][j] = 1 + min({dp[i-1][j], dp[i][j-1],
                  dp[i-1][j-1]});
      }
    }
  }
  return dp[m][n];
}
// Time: O(m*n), Space: O(m*n)

4.5 0/1 Knapsack Problem

Maximum value with weight constraint.

int knapsack(vector<int>& weights, vector<int>& values, int capacity) {
  int n = weights.size();
  vector<int> dp(capacity + 1, 0);
  
  for (int i = 0; i < n; i++) {
    for (int w = capacity; w >= weights[i]; w--) {
      dp[w] = max(dp[w], dp[w - weights[i]] + values[i]);
    }
  }
  return dp[capacity];
}
// Time: O(n*capacity), Space: O(capacity)

4.6 Word Break

Check if word can be segmented using words from dictionary.

bool wordBreak(string s, vector<string>& words) {
  unordered_set<string> dict(words.begin(), words.end());
  vector<bool> dp(s.size() + 1, false);
  dp[0] = true;
  
  for (int i = 1; i <= s.size(); i++) {
    for (int j = 0; j < i; j++) {
      if (dp[j] && dict.count(s.substr(j, i - j))) {
        dp[i] = true;
        break;
      }
    }
  }
  return dp[s.size()];
}
// Time: O(n^2 * m), Space: O(n) [m=avg word length]
